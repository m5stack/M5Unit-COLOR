103 inline static uint8_t linear(const uint16_t raw, const uint16_t low, const uint16_t high)
104 {
105 if (raw <= low) {
106 return 0U;
107 }
108 if (raw >= high) {
109 return 255U;
110 }
111 return static_cast<uint8_t>(
112 std::round((static_cast<float>(raw) - static_cast<float>(low)) / (static_cast<float>(high - low)) * 255.f));
113 }
114};
